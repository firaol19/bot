model MarketAnalysis {
  id        String   @id @default(uuid())
  botId     String
  bot       Bot      @relation(fields: [botId], references: [id])
  
  // Analysis metadata
  timestamp DateTime @default(now())
  decision  String   // SIGNAL_READY, WAITING_FOR_TREND, WAITING_FOR_SETUP, MONITORING_ENTRY, SCANNING, ERROR
  reason    String   @db.Text
  
  // Market data snapshot - stored as JSON for flexibility
  // Contains: timeframes (15m, 5m, 1m), indicators (EMAs, volumes), and decision details
  data      Json
  
  @@index([botId])
  @@index([timestamp])
}
